@layer components {

    /* 
       Override Tailwind Typography defaults 
       We must avoid :where() to ensure our specificity wins over the plugin's defaults.
    */
    /* 
       Override Tailwind Typography defaults 
       Targeting .editor-prose specifically to ensure our custom spacing wins.
    */
    .editor-prose {
        @apply prose prose-sm sm:prose-base focus:outline-none max-w-none;
    }

    /* 1. Remove Quotes */
    .editor-prose code::before,
    .editor-prose code::after,
    .editor-prose blockquote::before,
    .editor-prose blockquote::after {
        content: none !important;
    }

    /* 2. Tighten Spacing (Notion-like Density) */

    /* Paragraphs: ~6px gap (0.25em @ 16px = 4px, but collapsed margins logic implies ~8px visual gap) */
    .editor-prose p {
        margin-top: 0.25em;
        margin-bottom: 0.25em;
        line-height: 1.6;
    }

    /* Headings: Tight bottom margin to group with content */
    .editor-prose h1 {
        margin-top: 1em;
        /* ~32px */
        margin-bottom: 0.2em;
        /* ~6px */
        line-height: 1.2;
    }

    .editor-prose h2 {
        margin-top: 0.8em;
        /* ~24px */
        margin-bottom: 0.2em;
        /* ~6px */
        line-height: 1.3;
    }

    .editor-prose h3 {
        margin-top: 0.6em;
        /* ~18px */

        /* ~3px */

    }
